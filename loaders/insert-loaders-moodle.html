Clase 08 - Lógica de celdas y funciones esenciales
<div id="m08loader">Cargando la clase ...</div>
<script>
    (function() {
        var el = document.getElementById('m08loader');
        var m = location.href.match(/[?&]id=(\d+)/);
        var curso = m ? m[1] : null;
        var urlGuia = 'https://marcosxhernandez.github.io/unq-moodle/clases/Clase%2008%20-%20L%C3%B3gica%20de%20celdas%20y%20funciones%20esenciales.html';
        Promise.all([
            fetch('https://marcosxhernandez.github.io/unq-moodle/moodle/Clase%2008%20-%20L%C3%B3gica%20de%20celdas%20y%20funciones%20esenciales.html').then(function(r) { return r.text(); }),
            fetch('https://marcosxhernandez.github.io/unq-moodle/moodle/entregas.json').then(function(r) { return r.json(); })
        ]).then(function(resultados) {
            var html = resultados[0];
            var mapa = resultados[1][curso] || {};
            html = html.replace(/\{\{ENTREGA:(\d+):(\d+)\}\}/g, function(coincide, clase, orden) {
                return mapa[clase + ':' + orden] || '';
            });
            var doc = new DOMParser().parseFromString(html, 'text/html');
            el.innerHTML = doc.body.innerHTML;
        }).catch(function() {
            el.innerHTML = 'Error al sincronizar la clase. <a href="' + urlGuia + '" target="_blank" rel="noopener">Abrí la guía acá</a>.';
        });
    })();
</script>
